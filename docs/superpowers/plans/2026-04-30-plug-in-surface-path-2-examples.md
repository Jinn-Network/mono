## Cross-task conventions

- **Use the scaffolder.** Both examples bootstrap from `jinn create restorer ... --pattern=evaluator|alternative-harness`. The scaffolder already exists from Plan 1; this plan adds the *evaluator* and *alternative-harness* template trees the scaffolder reads from.

// src/oracle-stub.ts
export interface OracleSnapshot {
  intentId: string;
  resolvedOutcome: 0 | 1;            // 0 = NO, 1 = YES (binary markets)
  resolvedAt: number;                 // unix ms
}

export async function fetchResolution(intentId: string): Promise<OracleSnapshot> {
  // Deterministic stub keyed on intentId. Real impls call an on-chain oracle.
  const seed = Array.from(intentId).reduce((s, c) => (s * 31 + c.charCodeAt(0)) >>> 0, 0);
  return {
    intentId,
    resolvedOutcome: (seed & 1) as 0 | 1,
    resolvedAt: Date.now(),
  };
}

```ts
// src/score.ts
export function brier(p: number, outcome: 0 | 1): number {
  return (p - outcome) ** 2;
}

export interface ScoredForecast {
  p: number;
  outcome: 0 | 1;
}

export interface BrierDecomposition {
  reliability: number;
  resolution: number;
  uncertainty: number;
}

/**
 * Murphy 1973 decomposition: meanBrier = reliability - resolution + uncertainty.
 * Bins forecasts into 10 buckets by predicted probability.
 */
export function decomposeBrier(forecasts: readonly ScoredForecast[]): BrierDecomposition {
  if (forecasts.length === 0) {
    return { reliability: 0, resolution: 0, uncertainty: 0 };
  }

  const baseRate = forecasts.reduce((s, f) => s + f.outcome, 0) / forecasts.length;
  const uncertainty = baseRate * (1 - baseRate);

  const buckets = new Map<number, ScoredForecast[]>();
  for (const f of forecasts) {
    const bin = Math.min(9, Math.floor(f.p * 10));
    const list = buckets.get(bin);
    if (list) list.push(f);
    else buckets.set(bin, [f]);
  }

  let reliability = 0;
  let resolution = 0;
  for (const [, group] of buckets) {
    const n = group.length;
    const meanP = group.reduce((s, f) => s + f.p, 0) / n;
    const observedRate = group.reduce((s, f) => s + f.outcome, 0) / n;
    reliability += (n / forecasts.length) * (meanP - observedRate) ** 2;
    resolution += (n / forecasts.length) * (observedRate - baseRate) ** 2;
  }
  return { reliability, resolution, uncertainty };
}
```

import type {
  RestorerImpl,
  ExternalRestorerEnv,
  RestorationContext,
  RestorationOutput,
} from '@jinn-network/restorer-sdk';
import { brier } from './score.js';
import { fetchResolution } from './oracle-stub.js';

interface PredictionEnvelopeUnderEvaluation {
  forecastProbability: number;
  predictedOutcome?: 0 | 1;
}

export default function createEvaluator(env: ExternalRestorerEnv): RestorerImpl {
  return {
    name: env.implName,
    version: env.implVersion,
    supports({ kind, type }) {
      return kind === 'prediction.v0' && type === 'evaluation';
    },
    async isReady() {
      return env.stub ? { ready: false, reason: 'stub mode' } : { ready: true };
    },
    async run(ctx: RestorationContext): Promise<RestorationOutput> {
      const restoration = (ctx.intent.spec as Record<string, unknown>)?.restorationUnderEvaluation as
        | PredictionEnvelopeUnderEvaluation
        | undefined;
      if (!restoration || typeof restoration.forecastProbability !== 'number') {
        env.log({ level: 'error', msg: 'evaluator.no-restoration' });
        return {
          venueRef: { name: env.implName },
          gating: { score: null, error: 'missing restoration to score' },
          verdictPayload: { score: null, scoringRule: 'brier.v1', error: 'missing restoration' },
        };
      }
      const oracle = await fetchResolution(ctx.intent.id);
      const score = brier(restoration.forecastProbability, oracle.resolvedOutcome);
      env.log({
        level: 'info',
        msg: 'evaluator.scored',
        data: { intentId: ctx.intent.id, score, outcome: oracle.resolvedOutcome },
      });
      return {
        venueRef: { name: env.implName },
        gating: {
          score,
          scoringRule: 'brier.v1',
          resolvedOutcome: oracle.resolvedOutcome,
        },
        verdictPayload: {
          score,
          scoringRule: 'brier.v1',
          resolvedOutcome: oracle.resolvedOutcome,
          resolvedAt: oracle.resolvedAt,
        },
        informational: {
          forecastProbability: restoration.forecastProbability,
        },
      };
    },
  };
}
```

export async function runCoordinator({ ctx, harness }: CoordinatorArgs): Promise<RestorationOutput> {
  const orient = await runOrient({ ctx, harness });
  const strategy = await runStrategize({ ctx, harness, orient });
  const plan = await runPlan({ ctx, harness, strategy, orient });
  const execute = await runExecute({ ctx, harness, plan, strategy });
  const debrief = await runDebrief({ ctx, harness, execute, strategy });
  await runImprove({ ctx, harness, debrief });
  await runMemory({ ctx, harness, debrief });

  return {
    venueRef: { name: harness.name },
    gating: {
      timingPosture: strategy.timingPosture,
      executeReturnReason: execute.returnReason,
      debriefVerdict: debrief.successCriteriaMet,
    },
    informational: {
      strategyApproach: strategy.approach,
      planSteps: plan.steps.length,
    },
  };
}
```

import type { RestorationContext } from '@jinn-network/restorer-sdk';
import type { HarnessAdapter } from '../harness.js';
import type { OrientResult } from './orient.js';

export type TimingPosture = 'early-return' | 'hold-and-revise' | 'continuous-observation';

export interface StrategyResult {
  approach: string;
  successCriteria: string;
  timingPosture: TimingPosture;
}

export async function runStrategize(args: {
  ctx: RestorationContext;
  harness: HarnessAdapter;
  orient: OrientResult;
}): Promise<StrategyResult> {

    systemPrompt: 'You commit to one approach and freeze success criteria.',

The remaining five (`plan.ts`, `execute.ts`, `debrief.ts`, `improve.ts`, `memory.ts`) follow identical structure with their own input/output types — each is roughly 25 lines. **Each must be authored as a separate file with full content; do not stub them.** Per-file return shapes:

- `plan.ts` — `PlanResult { steps: Array<{ id: string; description: string }> }`
- `execute.ts` — `ExecuteResult { stepsCompleted: string[]; stepsFailed: string[]; returnReason: 'all-steps-completed' | 'time-budget-exhausted' | 'aborted' }`
- `debrief.ts` — `DebriefResult { successCriteriaMet: 'yes' | 'no' | 'partial'; rationale: string }`
- `improve.ts` — `ImproveResult { mutations: ReadonlyArray<{ path: string; description: string }> }` (for the example, returns empty mutations — real impls write to `ctx.implStateDir`)
- `memory.ts` — `MemoryResult { kept: number; pruned: number }` (example returns `{ kept: 0, pruned: 0 }`)

Each phase's `userPrompt` includes the prior phases' results as needed (matching the dependency graph in spec §3.3.3 reference). Budgets shrink phase-by-phase to leave Execute the largest budget.

/**

          return {
            approach: 'mock approach',
            successCriteria: 'always succeed',
            timingPosture: 'early-return',
          } as never;
        case 'plan':
          return { steps: [{ id: 's1', description: 'do the thing' }] } as never;
        case 'execute':

            stepsFailed: [],
            returnReason: 'all-steps-completed',

        case 'improve':
          return { mutations: [] } as never;
        case 'memory':

          throw new Error(`mock-harness: unknown promptId ${promptId}`);

    },
