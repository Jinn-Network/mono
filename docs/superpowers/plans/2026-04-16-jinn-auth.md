**Goal:** Add a `jinn auth` CLI verb that detects the operator's runtime context (container, Docker Compose, bare host), probes Claude's auth status, and guides login — plus integrate auth checks into `jinn run` preflight and `jinn doctor`.

**Architecture:** A shared preflight module (`src/preflight/claude-auth.ts`) handles context detection and auth probing via `claude auth status`. The CLI verb (`src/cli/commands/auth.ts`) wraps this with TTY-aware login flow. `main.ts` and `doctor.ts` consume the same probe.

**Tech Stack:** TypeScript, Node.js `child_process` (spawnSync), vitest for tests. No new dependencies.

```typescript
/**
 * Preflight check: is Claude authenticated?
 *
 * Detects the operator's runtime context (container, Docker Compose,
 * or bare host) and probes `claude auth status` in that context.
 */

import { existsSync, readFileSync } from 'node:fs';
import { spawnSync } from 'node:child_process';
import { join } from 'node:path';

export type AuthContext = 'container' | 'docker-compose' | 'bare';

export interface DetectContextOptions {
  cwd: string;
  dockerenvExists?: boolean;
  composeServiceExists?: boolean;
}

/**
 * Detect the runtime context for Claude authentication.
 *
 * Priority: container > docker-compose > bare.
 */
export function detectAuthContext(options: DetectContextOptions): AuthContext {
  if (options.dockerenvExists ?? existsSync('/.dockerenv')) {
    return 'container';
  }
  const hasComposeService = options.composeServiceExists ?? composeFileHasJinnDaemon(options.cwd);
  if (hasComposeService) {
    return 'docker-compose';
  }
  return 'bare';
}

function composeFileHasJinnDaemon(cwd: string): boolean {
  const composePath = join(cwd, 'docker-compose.yml');
  if (!existsSync(composePath)) return false;
  try {
    const content = readFileSync(composePath, 'utf8');
    return content.includes('jinn-daemon');
  } catch {
    return false;
  }
}
```

```typescript
export interface AuthProbeResult {
  authenticated: boolean;
  context: AuthContext;
  detail: string;
  email?: string;
}

interface SpawnResult {
  status: number | null;
  stdout: string;
  stderr: string;
}

export interface ProbeOptions {
  context: AuthContext;
  cwd: string;
  spawnResult?: SpawnResult;
}

/**
 * Probe Claude auth status. Accepts an optional pre-computed spawnResult
 * for testing; when omitted, spawns the real subprocess.
 */
export function probeClaudeAuth(options: ProbeOptions): AuthProbeResult {
  const { context, cwd } = options;
  const result = options.spawnResult ?? spawnAuthStatus(context, cwd);

  if (result.status !== 0) {
    return {
      authenticated: false,
      context,
      detail: 'claude auth status exited with non-zero status',
    };
  }

  try {
    const parsed = JSON.parse(result.stdout.trim());
    if (parsed.loggedIn === true) {
      return {
        authenticated: true,
        context,
        detail: parsed.email ? `Authenticated as ${parsed.email}` : 'Authenticated',
        email: parsed.email,
      };
    }
    return {
      authenticated: false,
      context,
      detail: 'Claude is not authenticated',
    };
  } catch {
    return {
      authenticated: false,
      context,
      detail: 'Could not parse claude auth status output',
    };
  }
}

function spawnAuthStatus(context: AuthContext, cwd: string): SpawnResult {
  if (context === 'docker-compose') {
    const result = spawnSync('docker', [
      'compose', '-f', join(cwd, 'docker-compose.yml'),
      'run', '--rm', '-T', '--no-deps',
      '--entrypoint', 'claude',
      'jinn-daemon', 'auth', 'status',
    ], { encoding: 'utf8', stdio: 'pipe', cwd });
    return {
      status: result.status ?? 1,
      stdout: result.stdout ?? '',
      stderr: result.stderr ?? '',
    };
  }
  // container or bare: claude is directly available
  const result = spawnSync('claude', ['auth', 'status'], {
    encoding: 'utf8',
    stdio: 'pipe',
    cwd,
  });
  return {
    status: result.status ?? 1,
    stdout: result.stdout ?? '',
    stderr: result.stderr ?? '',
  };
}
```

/**
 * Build the command array for `claude auth login` in the detected context.
 * Exported for use by the CLI verb (which needs to spawnSync with `inherit`
 * stdio for TTY pass-through).
 */
export function buildLoginCommand(context: AuthContext, cwd: string): { command: string; args: string[] } {
  if (context === 'docker-compose') {
    return {
      command: 'docker',
      args: [
        'compose', '-f', join(cwd, 'docker-compose.yml'),
        'run', '--rm', '-it', '--no-deps',
        '--entrypoint', 'claude',
        'jinn-daemon', 'auth', 'login',
      ],
    };
  }
  return { command: 'claude', args: ['auth', 'login'] };
}

```typescript
import { spawnSync } from 'node:child_process';
import { parseArgs } from 'node:util';
import type { CommandContext, CommandModule } from '../command.js';
import { emitResult } from '../output.js';
import { emitEnvelope } from '../../errors/envelope.js';
import {
  detectAuthContext,
  probeClaudeAuth,
  buildLoginCommand,
  type AuthContext,
} from '../../preflight/claude-auth.js';

const CONTEXT_LABELS: Record<AuthContext, string> = {
  container: 'inside this container',
  'docker-compose': 'inside the jinn-daemon Docker container',
  bare: 'on this machine',
};

async function run(ctx: CommandContext): Promise<void> {
  let parsed;
  try {
    parsed = parseArgs({
      args: ctx.argv,
      options: {
        json: { type: 'boolean', default: false },
        human: { type: 'boolean', default: false },
      },
      allowPositionals: false,
    });
  } catch (err) {
    emitEnvelope(
      {
        code: 'invalid_invocation',
        message: err instanceof Error ? err.message : String(err),
        exampleCli: 'jinn auth',
        details: { field: 'flags' },
      },
      { writer: ctx.writer, exit: ctx.exit },
    );
    return;
  }

  const cwd = process.cwd();
  const context = detectAuthContext({ cwd });
  const probe = probeClaudeAuth({ context, cwd });

  if (probe.authenticated) {
    emitResult(
      {
        schemaVersion: 1 as const,
        generatedAt: new Date().toISOString(),
        authenticated: true,
        context,
        detail: probe.detail,
        email: probe.email,
      },
      (v) => {
        const value = v as { detail: string; context: string };
        return `${value.detail} (${CONTEXT_LABELS[value.context as AuthContext]}).`;
      },
      {
        json: Boolean(parsed.values.json),
        human: Boolean(parsed.values.human),
        writer: ctx.writer,
        stdoutIsTty: ctx.stdoutIsTty,
        noColor: Boolean(ctx.env['NO_COLOR']),
      },
    );
    return;
  }

  // Not authenticated — need TTY for interactive login
  if (!ctx.stdoutIsTty) {
    emitEnvelope(
      {
        code: 'invalid_invocation',
        message: 'Claude is not authenticated. Run `jinn auth` in an interactive terminal.',
        hint: `Detected context: ${context}. Claude auth login requires an interactive session.`,
        exampleCli: 'jinn auth',
        details: { context, authenticated: false },
      },
      { writer: ctx.writer, exit: ctx.exit },
    );
    return;
  }

  // Interactive login
  const label = CONTEXT_LABELS[context];
  process.stderr.write(`Authenticating Claude ${label}...\n`);
  if (context === 'docker-compose') {
    process.stderr.write('A browser URL will appear — visit it to complete login.\n');
  }

  const loginCmd = buildLoginCommand(context, cwd);
  const loginResult = spawnSync(loginCmd.command, loginCmd.args, {
    stdio: 'inherit',
    cwd,
  });

  if ((loginResult.status ?? 1) !== 0) {
    emitEnvelope(
      {
        code: 'fatal',
        message: 'Claude auth login exited with a non-zero status.',
        hint: 'Try again with `jinn auth`.',
        details: { context, exitCode: loginResult.status },
      },
      { writer: ctx.writer, exit: ctx.exit },
    );
    return;
  }

  // Verify
  const verify = probeClaudeAuth({ context, cwd });
  if (!verify.authenticated) {
    emitEnvelope(
      {
        code: 'fatal',
        message: 'Login completed but verification failed. Claude does not report as authenticated.',
        hint: 'Try again with `jinn auth`.',
        details: { context },
      },
      { writer: ctx.writer, exit: ctx.exit },
    );
    return;
  }

  emitResult(
    {
      schemaVersion: 1 as const,
      generatedAt: new Date().toISOString(),
      authenticated: true,
      context,
      detail: verify.detail,
      email: verify.email,
    },
    (v) => {
      const value = v as { detail: string };
      return `Authenticated successfully. ${value.detail}`;
    },
    {
      json: Boolean(parsed.values.json),
      human: Boolean(parsed.values.human),
      writer: ctx.writer,
      stdoutIsTty: ctx.stdoutIsTty,
      noColor: Boolean(ctx.env['NO_COLOR']),
    },
  );
}

const command: CommandModule = {
  name: 'auth',
  summary: 'Authenticate Claude for this environment (Docker or host)',
  helpText: `Usage: jinn auth [--json] [--human]

Detects whether you are running inside a Docker container, via Docker
Compose, or on a bare host, then checks if Claude is authenticated in
that context.

If not authenticated and running in an interactive terminal, guides you
through \`claude auth login\`. In Docker Compose mode, this runs the
login inside the jinn-daemon container with the correct volume mounts.

If not authenticated and running non-interactively, exits with code 11
and a message directing you to run \`jinn auth\` in a terminal.

Context detection:
  container       /.dockerenv present (already inside Docker)
  docker-compose  docker-compose.yml with jinn-daemon in cwd
  bare            fallback — Claude CLI on the host

Examples:
  jinn auth
  jinn auth --json
  cd /path/to/client && jinn auth   # detects docker-compose.yml
`,
  run,
};

export default command;
```
